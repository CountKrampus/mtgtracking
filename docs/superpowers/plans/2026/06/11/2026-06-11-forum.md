> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a classic threaded forum (categories → thread list → thread view) with upvotes, full app integration (deck links, `[[Card]]` hover refs, playgroup tags), and admin moderation tools.

**Architecture:** Four new Mongoose models (ForumCategory, ForumThread, ForumPost, ForumMute) backed by a single `backend/routes/forum.js`. The frontend is a full-screen view at `/forum*` URLs — five new components under `frontend/src/components/Forum/` — wired into App.js route matching and the existing Sidebar. Two new admin tabs (Forum Categories, Forum Moderation) slot into AdminPanel's Community group.

**Tech Stack:** Node.js/Express/Mongoose, React (CRA), Tailwind CSS, Lucide icons, Jest + Supertest + mongodb-memory-server (backend tests).

```js
const express = require('express');
const router = express.Router();
const ForumCategory = require('../models/ForumCategory');
const ForumThread = require('../models/ForumThread');
const ForumPost = require('../models/ForumPost');
const ForumMute = require('../models/ForumMute');
const UserBan = require('../models/UserBan');
const { verifyToken, requireAuth, requireEditor, requireAdmin } = require('../middleware/auth');

// ── Helpers ───────────────────────────────────────────────────────────────────

function parseCardRefs(body) {
  const refs = [];
  const seen = new Set();
  const regex = /\[\[([^\]]+)\]\]/g;
  let m;
  while ((m = regex.exec(body)) !== null) {
    if (!seen.has(m[1])) { seen.add(m[1]); refs.push({ name: m[1] }); }
  }
  return refs;
}

function buildPostTree(flatPosts) {
  const map = {};
  flatPosts.forEach(p => {
    const obj = p.toObject ? p.toObject() : { ...p };
    map[obj._id.toString()] = { ...obj, replies: [], _depth: 1 };
  });
  const roots = [];
  const sorted = flatPosts.slice().sort((a, b) => new Date(a.createdAt) - new Date(b.createdAt));
  sorted.forEach(p => {
    const node = map[p._id.toString()];
    const parentId = p.parentPostId?.toString();
    const parent = parentId ? map[parentId] : null;
    if (!parent || parent._depth >= 3) {
      roots.push(node);
    } else {
      node._depth = parent._depth + 1;
      parent.replies.push(node);
    }
  });
  return roots;
}

async function checkForumAccess(req, res, next) {
  const uid = req.user._id;
  if (await UserBan.isUserBanned(uid)) return res.status(403).json({ message: 'You are banned from this site' });
  if (await ForumMute.isUserMuted(uid)) return res.status(403).json({ message: 'You are muted from the forum' });
  next();
}

// ── Public routes ─────────────────────────────────────────────────────────────

// GET /api/forum/categories
router.get('/categories', verifyToken, async (req, res) => {
  try {
    const cats = await ForumCategory.find().sort({ order: 1, name: 1 }).lean();
    const withCounts = await Promise.all(cats.map(async cat => ({
      ...cat,
      threadCount: await ForumThread.countDocuments({ categoryId: cat._id }),
      lastThread: await ForumThread.findOne({ categoryId: cat._id })
        .sort({ lastReplyAt: -1 }).select('title lastReplyAt authorUsername').lean()
    })));
    res.json(withCounts);
  } catch (e) { res.status(500).json({ message: e.message }); }
});

// GET /api/forum/categories/:slug/threads
router.get('/categories/:slug/threads', verifyToken, async (req, res) => {
  try {
    const cat = await ForumCategory.findOne({ slug: req.params.slug }).lean();
    if (!cat) return res.status(404).json({ message: 'Category not found' });

    const sort = req.query.sort || 'latest';
    const page = Math.max(1, parseInt(req.query.page) || 1);
    const limit = 20;
    const skip = (page - 1) * limit;

    let sortObj;
    if (sort === 'new') sortObj = { createdAt: -1 };
    else if (sort === 'hot') sortObj = { 'upvotes.length': -1, createdAt: -1 };
    else sortObj = { isPinned: -1, lastReplyAt: -1 };

    const [threads, total] = await Promise.all([
      ForumThread.find({ categoryId: cat._id }).sort(sortObj).skip(skip).limit(limit).lean(),
      ForumThread.countDocuments({ categoryId: cat._id })
    ]);

    const userId = req.user?._id?.toString();
    const result = threads.map(t => ({
      ...t,
      upvoteCount: t.upvotes.length,
      hasUpvoted: userId ? t.upvotes.some(id => id.toString() === userId) : false,
      upvotes: undefined
    }));

    res.json({ category: cat, threads: result, pagination: { page, limit, total, pages: Math.ceil(total / limit) } });
  } catch (e) { res.status(500).json({ message: e.message }); }
});

// GET /api/forum/threads/:id
router.get('/threads/:id', verifyToken, async (req, res) => {
  try {
    const thread = await ForumThread.findById(req.params.id).lean();
    if (!thread) return res.status(404).json({ message: 'Thread not found' });

    await ForumThread.findByIdAndUpdate(req.params.id, { $inc: { viewCount: 1 } });

    const posts = await ForumPost.find({ threadId: thread._id }).sort({ createdAt: 1 }).lean();
    const userId = req.user?._id?.toString();

    const postsWithMeta = posts.map(p => ({
      ...p,
      upvoteCount: p.upvotes.length,
      hasUpvoted: userId ? p.upvotes.some(id => id.toString() === userId) : false,
      upvotes: undefined
    }));

    res.json({
      thread: {
        ...thread,
        upvoteCount: thread.upvotes.length,
        hasUpvoted: userId ? thread.upvotes.some(id => id.toString() === userId) : false,
        upvotes: undefined
      },
      posts: buildPostTree(postsWithMeta)
    });
  } catch (e) { res.status(500).json({ message: e.message }); }
});

module.exports = router;
module.exports.buildPostTree = buildPostTree;
```

```js
// ── Authenticated — threads ───────────────────────────────────────────────────

// POST /api/forum/threads
router.post('/threads', verifyToken, requireAuth, requireEditor, checkForumAccess, async (req, res) => {
  try {
    const { categoryId, title, body, deckRef, playgroupRef } = req.body;
    if (!categoryId || !title?.trim() || !body?.trim())
      return res.status(400).json({ message: 'categoryId, title, and body are required' });

    const cat = await ForumCategory.findById(categoryId);
    if (!cat) return res.status(404).json({ message: 'Category not found' });

    const thread = await ForumThread.create({
      categoryId,
      authorId: req.user._id,
      authorUsername: req.user.username,
      authorDisplayName: req.user.displayName || req.user.username,
      title: title.trim(),
      body: body.trim(),
      cardRefs: parseCardRefs(body),
      deckRef: deckRef || undefined,
      playgroupRef: playgroupRef || undefined
    });
    res.status(201).json(thread);
  } catch (e) { res.status(400).json({ message: e.message }); }
});

// PUT /api/forum/threads/:id
router.put('/threads/:id', verifyToken, requireAuth, checkForumAccess, async (req, res) => {
  try {
    const thread = await ForumThread.findById(req.params.id);
    if (!thread) return res.status(404).json({ message: 'Thread not found' });
    if (thread.authorId.toString() !== req.user._id.toString() && req.user.role !== 'admin')
      return res.status(403).json({ message: 'Not authorized' });
    if (thread.isLocked && req.user.role !== 'admin')
      return res.status(403).json({ message: 'Thread is locked' });

    const { title, body, deckRef, playgroupRef } = req.body;
    if (title) thread.title = title.trim();
    if (body) { thread.body = body.trim(); thread.cardRefs = parseCardRefs(body); }
    if (deckRef !== undefined) thread.deckRef = deckRef;
    if (playgroupRef !== undefined) thread.playgroupRef = playgroupRef;
    await thread.save();
    res.json(thread);
  } catch (e) { res.status(400).json({ message: e.message }); }
});

// DELETE /api/forum/threads/:id — cascades to posts
router.delete('/threads/:id', verifyToken, requireAuth, async (req, res) => {
  try {
    const thread = await ForumThread.findById(req.params.id);
    if (!thread) return res.status(404).json({ message: 'Thread not found' });
    if (thread.authorId.toString() !== req.user._id.toString() && req.user.role !== 'admin')
      return res.status(403).json({ message: 'Not authorized' });

    await ForumPost.deleteMany({ threadId: thread._id });
    await thread.deleteOne();
    res.json({ message: 'Thread deleted' });
```

```jsx
import React, { useState, useEffect } from 'react';
import axios from 'axios';
import { Plus, Trash2 } from 'lucide-react';

const API_URL = process.env.REACT_APP_API_URL || `http://${window.location.hostname}:5000/api`;

export default function ForumModerationTab() {
  const [mutes, setMutes] = useState([]);
  const [loading, setLoading] = useState(true);
  const [form, setForm] = useState({ username: '', reason: '', expiresAt: '' });
  const [error, setError] = useState('');
  const [saving, setSaving] = useState(false);

  const load = () => {
    setLoading(true);
    axios.get(`${API_URL}/forum/mutes`).then(r => setMutes(r.data)).catch(() => {}).finally(() => setLoading(false));
  };

  useEffect(() => { load(); }, []);

  const handleMute = async () => {
    setError('');
    if (!form.username.trim() || !form.reason.trim()) return setError('Username and reason are required');
    setSaving(true);
    try {
      // Look up userId by username first
      const usersRes = await axios.get(`${API_URL}/admin/users`, { params: { search: form.username, limit: 1 } });
      const user = usersRes.data.users?.[0];
      if (!user) return setError('User not found');
      await axios.post(`${API_URL}/forum/mutes`, {
        userId: user._id,
        reason: form.reason,
        expiresAt: form.expiresAt || null
      });
      setForm({ username: '', reason: '', expiresAt: '' });
      load();
    } catch (e) { setError(e.response?.data?.message || 'Failed to mute user'); }
    finally { setSaving(false); }
  };

  const handleUnmute = async (userId, username) => {
    if (!window.confirm(`Unmute ${username}?`)) return;
    try { await axios.delete(`${API_URL}/forum/mutes/${userId}`); load(); }
    catch (e) { alert(e.response?.data?.message || 'Failed to unmute'); }
  };

  return (
    <div className="p-4">
      <h3 className="text-white font-semibold mb-4">Forum Mutes</h3>
      <p className="text-white/40 text-sm mb-4">Muted users can read threads but cannot post or reply. For full site bans, use the Bans tab.</p>

      {/* Mute form */}
      <div className="bg-gray-800/40 border border-white/10 rounded-lg p-4 mb-5">
        <p className="text-xs text-white/40 uppercase tracking-wider mb-3">Mute a User</p>
        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 mb-3">
          <input value={form.username} onChange={e => setForm(f => ({ ...f, username: e.target.value }))}
            placeholder="Username" className="bg-gray-700 border border-white/10 rounded-lg px-3 py-2 text-white text-sm" />
          <input value={form.reason} onChange={e => setForm(f => ({ ...f, reason: e.target.value }))}
            placeholder="Reason" className="bg-gray-700 border border-white/10 rounded-lg px-3 py-2 text-white text-sm" />
          <div>
            <input type="datetime-local" value={form.expiresAt} onChange={e => setForm(f => ({ ...f, expiresAt: e.target.value }))}
              className="w-full bg-gray-700 border border-white/10 rounded-lg px-3 py-2 text-white text-sm" />
            <p className="text-white/30 text-xs mt-1">Leave blank for permanent mute</p>
          </div>
        </div>
        {error && <p className="text-red-400 text-sm mb-2">{error}</p>}
        <button onClick={handleMute} disabled={saving}
          className="flex items-center gap-1.5 bg-orange-600 hover:bg-orange-700 disabled:opacity-50 text-white px-4 py-2 rounded-lg text-sm font-medium transition">
          <Plus size={14} /> {saving ? 'Muting…' : 'Mute User'}
        </button>
      </div>

      {/* Active mutes table */}
      <div className="border border-white/10 rounded-xl overflow-hidden">
        <div className="grid grid-cols-[1fr_2fr_120px_80px] px-4 py-2 bg-white/5 text-xs text-white/30 uppercase tracking-wider">
          <span>User</span><span>Reason</span><span>Expires</span><span>Action</span>
        </div>
        {loading ? (
          <div className="py-8 text-center text-white/40 text-sm">Loading…</div>
        ) : mutes.length === 0 ? (
          <div className="py-8 text-center text-white/40 text-sm">No active forum mutes</div>
        ) : (
          mutes.map(m => (
            <div key={m._id} className="grid grid-cols-[1fr_2fr_120px_80px] items-center px-4 py-3 border-t border-white/6 hover:bg-white/3 text-sm">
              <span className="text-white font-medium">{m.userId?.username || m.userId}</span>
              <span className="text-white/60">{m.reason}</span>
              <span className="text-white/40 text-xs">{m.expiresAt ? new Date(m.expiresAt).toLocaleDateString() : 'Permanent'}</span>
              <button onClick={() => handleUnmute(m.userId?._id || m.userId, m.userId?.username)}
                className="flex items-center gap-1 text-xs text-white/40 hover:text-red-400 transition">
                <Trash2 size={13} /> Unmute
              </button>
            </div>
          ))
        )}
      </div>
    </div>
  );
}
```
